Live Express Backend Server Code

Full standalone Node.js script handling OAuth, STK Push, Webhook Callback, and Router authorization.

const express = require('express');
const axios = require('axios');
const cors = require('cors');

const app = express();
app.use(express.json());
app.use(cors());

// PRE-FILLED LIVE SAFARICOM DARAJA CREDENTIALS
const CONSUMER_KEY = 'AGJGJGGktfRf8OHhos6TJXKupDMb3f2wG9pZDdF8tG5qm0qK';
const CONSUMER_SECRET = 'xmd2sXJzRA6Chl1KHGvR9jG0qtSBse1oRBCdgBjb4DnO7SEi4QzPVwKNWWraubJd';
const PASSKEY = '103f80194fedeaefb352960fb340d766ef3f84a9dcb96aafdd21d0b0bb8dfbab';
const SHORTCODE = '174379'; // Paybill or Till Shortcode

// 1. GET DARAJA OAUTH ACCESS TOKEN
async function getAccessToken() {
    const auth = Buffer.from(`${CONSUMER_KEY}:${CONSUMER_SECRET}`).toString('base64');
    const url = 'https://sandbox.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials';
    const response = await axios.get(url, { headers: { Authorization: `Basic ${auth}` } });
    return response.data.access_token;
}

// 2. TRIGGER DARAJA STK PUSH
app.post('/api/stkpush', async (req, res) => {
    try {
        const { phone, amount, callbackUrl } = req.body;
        const token = await getAccessToken();
        const timestamp = new Date().toISOString().replace(/[^0-9]/g, '').slice(0, 14);
        const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString('base64');

        const formattedPhone = phone.replace(/^0/, '254');
        const payload = {
            BusinessShortCode: SHORTCODE,
            Password: password,
            Timestamp: timestamp,
            TransactionType: 'CustomerPayBillOnline',
            Amount: Math.round(amount),
            PartyA: formattedPhone,
            PartyB: SHORTCODE,
            PhoneNumber: formattedPhone,
            CallBackURL: callbackUrl || 'https://lin13423.github.io/Billing-system/callback',
            AccountReference: 'LinWifi',
            TransactionDesc: 'Internet Access Purchase'
        };

        const response = await axios.post(
            'https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest',
            payload,
            { headers: { Authorization: `Bearer ${token}` } }
        );

        res.json({ success: true, data: response.data });
    } catch (error) {
        res.status(500).json({ success: false, error: error.response?.data || error.message });
    }
});

// 3. INSTANT CALLBACK WEBHOOK ENDPOINT
app.post('/api/callback', (req, res) => {
    const callbackData = req.body.Body.stkCallback;
    console.log('Received M-Pesa Callback:', JSON.stringify(callbackData, null, 2));

    if (callbackData.ResultCode === 0) {
        // SUCCESS: Authorize device on router gateway
        const items = callbackData.CallbackMetadata.Item;
        const amount = items.find(i => i.Name === 'Amount')?.Value;
        const receipt = items.find(i => i.Name === 'MpesaReceiptNumber')?.Value;
        const phone = items.find(i => i.Name === 'PhoneNumber')?.Value;

        console.log(`[ROUTER AUTH] Device Paid ${amount} KES via ${receipt} (${phone}). Session Unlocked!`);
    } else {
        console.log(`[STK CANCEL] Transaction failed with code ${callbackData.ResultCode}`);
    }

    res.json({ ResultCode: 0, ResultDesc: 'Accepted' });
});

app.listen(3000, () => console.log('Lin Billing Gateway Server running on port 3000'));
